// ========================================================================= // 1. FUNZIONE DI NORMALIZZAZIONE (Retrocompatibilità - Protegge i vecchi dati) // ========================================================================= function normalizzaEvento(ev) { if (!ev) ev = {}; return { id: ev.id || '', nomeEvento: ev.nomeEvento || 'Evento senza nome', sala: ev.sala || 'Non specificata', tavoli: ev.tavoli || '-', // Retrocompatibilità: usa ospitiAdulti se presente, altrimenti il vecchio campo 'ospiti' ospitiAdulti: Number(ev.ospitiAdulti !== undefined ? ev.ospitiAdulti : (ev.ospiti || 0)), prezzoAdulto: Number(ev.prezzoAdulto !== undefined ? ev.prezzoAdulto : (ev.prezzo || 0)), menuAdulti: ev.menuAdulti || ev.menu || 'Non specificato', // Nuovi campi dedicati ai bambini ospitiBambini: Number(ev.ospitiBambini || 0), prezzoBambino: Number(ev.prezzoBambino || 0), menuBambini: ev.menuBambini || '', addobbiColori: ev.addobbiColori || '-', noteAllergie: ev.noteAllergie || '-' }; } // ========================================================================= // 2. GENERAZIONE SCHEDA EVENTO (Interfaccia Principale) // NO Totale € e NO Prezzi a persona. Mostra Menu Adulti e Menu Bimbi. // ========================================================================= function generaHtmlCardEvento(eventoGrezzo) { const ev = normalizzaEvento(eventoGrezzo); // Costruzione stringa ospiti let testoOspiti = `👥 ${ev.ospitiAdulti} Adulti`; if (ev.ospitiBambini > 0) { testoOspiti += ` + ${ev.ospitiBambini} Bimbi`; } return `

🎉 ${ev.nomeEvento}

📍 ${ev.sala} — ${testoOspiti} (Tavoli: ${ev.tavoli})

🍽️ Menu Adulti: ${ev.menuAdulti}

${(ev.ospitiBambini > 0 || ev.menuBambini) ? `

🎈 Menu Bimbi: ${ev.menuBambini || 'Menu Standard Bimbi'}

` : ''}

🎨 Addobbi/Colori: ${ev.addobbiColori}

⚠️ Note/Allergie: ${ev.noteAllergie}

`; } // ========================================================================= // 3. FORM NUOVO / MODIFICA EVENTO (Finestra / Modal) // ========================================================================= function generaHtmlFormEvento(eventoGrezzo = {}) { const ev = normalizzaEvento(eventoGrezzo); return `

${ev.id ? '✏️ Modifica Evento' : '➕ Nuovo Evento'}

👥 Adulti

🎈 Bambini

`; } // ========================================================================= // 4. ESTRAZIONE DATI PER IL SALVATAGGIO IN FIREBASE // Mantiene anche i vecchi campi 'ospiti' e 'prezzo' per compatibilità // ========================================================================= function estraiDatiDaForm() { const ospitiAdulti = Number(document.getElementById('ev_ospitiAdulti').value) || 0; const ospitiBambini = Number(document.getElementById('ev_ospitiBambini').value) || 0; const prezzoAdulto = Number(document.getElementById('ev_prezzoAdulto').value) || 0; const prezzoBambino = Number(document.getElementById('ev_prezzoBambino').value) || 0; return { nomeEvento: document.getElementById('ev_nomeEvento').value.trim(), sala: document.getElementById('ev_sala').value.trim(), tavoli: document.getElementById('ev_tavoli').value.trim(), // Nuovi campi separati ospitiAdulti: ospitiAdulti, prezzoAdulto: prezzoAdulto, menuAdulti: document.getElementById('ev_menuAdulti').value.trim(), ospitiBambini: ospitiBambini, prezzoBambino: prezzoBambino, menuBambini: document.getElementById('ev_menuBambini').value.trim(), addobbiColori: document.getElementById('ev_addobbiColori').value.trim(), noteAllergie: document.getElementById('ev_noteAllergie').value.trim(), // CAMPI DUALI DI SICUREZZA (Impediscono perdite di dati su vecchi report) ospiti: ospitiAdulti + ospitiBambini, prezzo: prezzoAdulto }; } // ========================================================================= // 5. STAMPA / STORICO (Sotto Chiave Master) // I prezzi e il totale compaiono SOLO quando si attiva la Chiave Master o in Stampa // ========================================================================= function generaHtmlStampaEvento(eventoGrezzo, isChiaveMasterAttiva = false) { const ev = normalizzaEvento(eventoGrezzo); const totaleAdulti = ev.ospitiAdulti * ev.prezzoAdulto; const totaleBambini = ev.ospitiBambini * ev.prezzoBambino; const totaleGenerale = totaleAdulti + totaleBambini; return `

${ev.nomeEvento} — ${ev.sala} (Tavoli: ${ev.tavoli})

Adulti: ${ev.ospitiAdulti} px | Menu Adulti: ${ev.menuAdulti}

${ev.ospitiBambini > 0 ? `

Bambini: ${ev.ospitiBambini} px | Menu Bimbi: ${ev.menuBambini}

` : ''}

Addobbi: ${ev.addobbiColori} | Note/Allergie: ${ev.noteAllergie}

${isChiaveMasterAttiva ? `

💰 Adulti: ${ev.ospitiAdulti} × ${ev.prezzoAdulto} € = ${totaleAdulti} €

${ev.ospitiBambini > 0 ? `

💰 Bambini: ${ev.ospitiBambini} × ${ev.prezzoBambino} € = ${totaleBambini} €

` : ''}

💶 TOTALE EVENTO: ${totaleGenerale} €

` : `

🔒 Dettagli economici visibili sbloccando la Chiave Master.

`}
`; }
